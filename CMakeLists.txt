cmake_minimum_required(VERSION 3.22.1)
project(perekup_native LANGUAGES C CXX)
add_library(perekup SHARED main.cpp)
find_library(log-lib log)
find_library(android-lib android)
find_library(gles-lib GLESv2)
find_library(egl-lib EGL)
target_link_libraries(perekup ${log-lib} ${android-lib} ${gles-lib} ${egl-lib})
set(NATIVE_APP_GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(native_app_glue STATIC ${NATIVE_APP_GLUE}/android_native_app_glue.c)
target_include_directories(perekup PRIVATE ${NATIVE_APP_GLUE})
target_link_libraries(perekup native_app_glue)
